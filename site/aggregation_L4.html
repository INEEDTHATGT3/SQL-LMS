Materialized Views & Continuous Aggregates

Materialized Views — Pre-computed Snapshots

materialized view
-- Create materialized view
CREATE MATERIALIZED VIEW dept_stats AS
SELECT dept_id,
  COUNT(*) AS emp_count,
  AVG(salary) AS avg_salary,
  SUM(salary) AS total_salary
FROM employees
GROUP BY dept_id;

-- Refresh (full rebuild)
REFRESH MATERIALIZED VIEW dept_stats;

-- Refresh CONCURRENTLY (zero-downtime)
REFRESH MATERIALIZED VIEW CONCURRENTLY dept_stats;
-- Requires: unique index on materialized view
CREATE UNIQUE INDEX idx_dept_stats_dept ON dept_stats (dept_id);

Continuous Aggregates (TimescaleDB)

continuous aggregate
-- TimescaleDB continuous aggregate
CREATE MATERIALIZED VIEW hourly_events
WITH (timescaledb.continuous) AS
SELECT time_bucket('1 hour', time) AS bucket,
  device_id,
  COUNT(*) AS event_count,
  AVG(value) AS avg_value
FROM events
GROUP BY bucket, device_id;

-- Automatic refresh (background)
-- Add refresh policy
SELECT add_continuous_aggregate_policy('hourly_events',
  start_offset => INTERVAL '3 hours',
  end_offset => INTERVAL '1 hour',
  schedule_interval => INTERVAL '1 hour');

-- Real-time view (combines materialized + raw)
SELECT * FROM hourly_events WHERE bucket > now() - INTERVAL '1 day';

Incremental Aggregate Maintenance

incremental aggregate pattern
-- Table for incremental aggregates
CREATE TABLE daily_sales_agg (
  sale_date DATE PRIMARY KEY,
  total_sales NUMERIC,
  order_count BIGINT
);

-- Incremental refresh (daily)
INSERT INTO daily_sales_agg (sale_date, total_sales, order_count)
SELECT sale_date, SUM(amount), COUNT(*)
FROM orders
WHERE sale_date = CURRENT_DATE - 1
GROUP BY sale_date
ON CONFLICT (sale_date) DO UPDATE SET
  total_sales = EXCLUDED.total_sales,
  order_count = EXCLUDED.order_count;

continuous aggregateTimescaleDB: automatically maintained materialized view with incremental refresh.
