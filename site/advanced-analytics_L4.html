Custom Window Aggregates (PostgreSQL 14+)

CREATE AGGREGATE ... WINDOW enables custom window aggregates with moving window support. Define state transition, final function, and optional moving window functions (msfunc, minvfunc) for O(1) sliding.
custom window aggregate
-- Custom moving average aggregate
CREATE FUNCTION mov_avg_sfunc(state internal, val numeric, window_start int, window_end int)
RETURNS internal LANGUAGE C AS '...';

CREATE FUNCTION mov_avg_final(state internal)
RETURNS numeric LANGUAGE C AS '...';

-- Moving window functions for O(1) sliding
CREATE FUNCTION mov_avg_msfunc(state internal, val numeric, window_start int, window_end int)
RETURNS internal LANGUAGE C AS '...';

CREATE FUNCTION mov_avg_minvfunc(state internal, val numeric, window_start int, window_end int)
RETURNS internal LANGUAGE C AS '...';

CREATE AGGREGATE mov_avg(numeric) (
  SFUNC = mov_avg_sfunc,
  STYPE = internal,
  FINALFUNC = mov_avg_final,
  MSFUNC = mov_avg_msfunc,
  MINVFUNC = mov_avg_minvfunc,
  WINDOW = true
);

-- Usage
SELECT mov_avg(salary) OVER (ORDER BY hire_date ROWS BETWEEN 2 PRECEDING AND CURRENT ROW)
FROM employees;

IGNORE NULLS & Advanced NULL Handling

IGNORE NULLS / RESPECT NULLS (PostgreSQL 15+)

ignore nulls
-- LAG/LEAD with IGNORE NULLS
SELECT 
  sale_date,
  amount,
  LAG(amount) IGNORE NULLS OVER (ORDER BY sale_date) AS prev_non_null,
  LEAD(amount) IGNORE NULLS OVER (ORDER BY sale_date) AS next_non_null
FROM daily_sales;

-- FIRST_VALUE / LAST_VALUE with IGNORE NULLS
SELECT 
  sale_date,
  amount,
  FIRST_VALUE(amount) IGNORE NULLS OVER (ORDER BY sale_date) AS first_sale,
  LAST_VALUE(amount) IGNORE NULLS OVER (
    ORDER BY sale_date
    ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING
  ) AS last_sale
FROM daily_sales;

Window Function Internals — WindowAgg, Sort, Spilling

WindowAgg Execution Model

PostgreSQL WindowAgg node: 1) Sort input by PARTITION BY + ORDER BY, 2) For each partition, evaluate window functions in sequence, 3) For each row, compute frame boundaries and evaluate functions. Multiple functions with same PARTITION+ORDER share sort and single WindowAgg pass.

Triggers & Transaction Control — Audit, Constraints, Cascades

CREATE TRIGGER — BEFORE / AFTER / INSTEAD OF

audit triggers
-- AFTER ROW: capture who changed what
CREATE TABLE emp_audit (
  op      text NOT NULL,
  emp_id  int  NOT NULL,
  changed_at timestamptz NOT NULL DEFAULT now(),
  changed_by text NOT NULL DEFAULT current_user
);

CREATE FUNCTION log_emp_change() RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO emp_audit VALUES (TG_OP, NEW.id);
  RETURN NEW;
END;
$$;

CREATE TRIGGER emp_audit_ai
AFTER INSERT OR UPDATE ON employees
FOR EACH ROW EXECUTE FUNCTION log_emp_change();

INSTEAD OF on views

instead of trigger
-- Make a view writable via INSTEAD OF
CREATE VIEW emp_full AS
SELECT e.id, e.name, d.name AS dept
FROM employees e JOIN departments d ON d.id = e.dept_id;

CREATE FUNCTION emp_full_insert() RETURNS trigger
LANGUAGE plpgsql AS $$
BEGIN
  INSERT INTO departments(name) VALUES (NEW.dept)
    ON CONFLICT (name) DO NOTHING;
  INSERT INTO employees(id, name, dept_id)
    SELECT NEW.id, NEW.name, id FROM departments WHERE name = NEW.dept;
  RETURN NEW;
END;
$$;

CREATE TRIGGER emp_full_ioi
INSTEAD OF INSERT ON emp_full
FOR EACH ROW EXECUTE FUNCTION emp_full_insert();

Constraint enforcement & audit hygiene

validation trigger
-- Cross-table invariant a plain FK can't express
CREATE FUNCTION check_room_capacity() RETURNS trigger
LANGUAGE plpgsql AS $$
DECLARE cap int;
BEGIN
  SELECT capacity INTO cap FROM rooms WHERE id = NEW.room_id;
  IF (SELECT COUNT(*) FROM bookings
       WHERE room_id = NEW.room_id
         AND tstzrange(date, date + duration) &&
             tstzrange(NEW.date, NEW.date + NEW.duration)) >= cap
  THEN
    RAISE EXCEPTION 'room % is over capacity', NEW.room_id;
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER bookings_cap_before
BEFORE INSERT OR UPDATE ON bookings
FOR EACH ROW EXECUTE FUNCTION check_room_capacity();
