Advanced PIVOT — Dynamic Columns & Multi-Level

Static PIVOT works when categories are known. Dynamic PIVOT generates columns at runtime. Multi-level PIVOT (e.g., year → quarter → month) requires chaining or dynamic SQL. PostgreSQL: dynamic crosstab via execute format().
dynamic crosstab
-- Dynamic columns: use EXECUTE format()
DO $$
DECLARE
  cols TEXT;
BEGIN
  SELECT string_agg(DISTINCT quote_ident(quarter), ', ') INTO cols
  FROM sales;
  EXECUTE format(
    'SELECT * FROM crosstab(
      ''SELECT dept, quarter, SUM(sales) FROM sales GROUP BY dept, quarter ORDER BY 1,2'',
      ''SELECT DISTINCT quarter FROM sales ORDER BY 1''
    ) AS ct(dept TEXT, %s)',
    cols
  );
END $$;

Advanced LATERAL — Top-N, Correlated, Parallel

LATERAL for Top-N Per Group

lateral top-n per group
-- Top 3 products per category by sales
SELECT c.name, p.name, p.sales
FROM categories c
LEFT JOIN LATERAL (
  SELECT name, sales
  FROM products p
  WHERE p.category_id = c.id
  ORDER BY p.sales DESC
  LIMIT 3
) p ON true;

-- With ROW_NUMBER alternative (no LATERAL)
WITH ranked AS (
  SELECT p.*, ROW_NUMBER() OVER (PARTITION BY category_id ORDER BY sales DESC) AS rn
  FROM products
)
SELECT * FROM ranked WHERE rn <= 3;

LATERAL with Aggregate Subqueries

lateral + aggregate
-- For each customer: total orders, avg order value, latest order date
SELECT c.name,
  COALESCE(oa.total_orders, 0) AS total_orders,
  oa.avg_order,
  oa.latest_order
FROM customers c
LEFT JOIN LATERAL (
  SELECT 
    COUNT(*) AS total_orders,
    AVG(total) AS avg_order,
    MAX(order_date) AS latest_order
  FROM orders o
  WHERE o.customer_id = c.id
) oa ON true;

Advanced Frames & Sampling

GROUPS Frame (PostgreSQL 14+)

groups frame
-- GROUPS: counts groups of peer rows (same ORDER BY value)
-- 1 group = all rows with same ORDER BY value
SELECT 
  salary,
  COUNT(*) OVER (
    ORDER BY salary
    GROUPS BETWEEN 1 PRECEDING AND 1 FOLLOWING
  ) AS peer_count
FROM employees;

Advanced Sampling — SYSTEM vs BERNOULLI

sampling methods
-- SYSTEM: page-level (fast, approximate)
SELECT * FROM large_table TABLESAMPLE SYSTEM (10);

-- BERNOULLI: row-level (accurate %, slower)
SELECT * FROM large_table TABLESAMPLE BERNOULLI (10);

-- REPEATABLE for deterministic results
SELECT * FROM large_table TABLESAMPLE SYSTEM (10) REPEATABLE (42);

-- With WHERE (pushdown may not work)
SELECT * FROM large_table TABLESAMPLE SYSTEM (10) WHERE status = 'active';

SYSTEM sampling is fast but can be skewed if data clustered on pages. BERNOULLI is statistically sound but scans all rows. REPEATABLE(seed) for reproducible samples.
