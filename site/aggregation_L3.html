FILTER Clause — Conditional Aggregation

FILTER = conditional aggregate without CASE. One scan, multiple aggregates with different conditions. Cleaner than SUM(CASE...), optimizer-friendly, index-friendly.
filter clause
-- Count active employees per dept
SELECT dept_id,
  COUNT(*) FILTER (WHERE status = 'active') AS active_cnt,
  COUNT(*) FILTER (WHERE status = 'pending') AS pending_cnt,
  COUNT(*) FILTER (WHERE status = 'inactive') AS inactive_cnt
FROM employees
GROUP BY dept_id;

-- Equivalent CASE (verbose, less readable)
SELECT dept_id,
  SUM(CASE WHEN status = 'active' THEN 1 ELSE 0 END) AS active_cnt,
  SUM(CASE WHEN status = 'pending' THEN 1 ELSE 0 END) AS pending_cnt
FROM employees GROUP BY dept_id;

-- Multiple aggregates, different filters
SELECT dept_id,
  AVG(salary) FILTER (WHERE status = 'active') AS avg_active,
  AVG(salary) FILTER (WHERE status = 'pending') AS avg_pending
FROM employees GROUP BY dept_id;

Approximate Aggregates — HyperLogLog, Percentile

Approximate Count Distinct

approx_count_distinct
-- Exact (slow on large tables)
SELECT COUNT(DISTINCT user_id) FROM events;

-- Approximate (fast, ~1-2% error)
SELECT approx_count_distinct(user_id) FROM events;

-- With GROUP BY
SELECT dept_id, approx_count_distinct(user_id)
FROM events
GROUP BY dept_id;

-- Precision control (PostgreSQL: hyperloglog extension)
SELECT approx_count_distinct(user_id, 12) FROM events;
-- 12 = log2m parameter, higher = more accurate, more memory

Percentile Aggregates

percentile functions
-- Continuous percentile (interpolated)
SELECT percentile_cont(0.5) WITHIN GROUP (ORDER BY salary) FROM employees;

-- Discrete percentile (actual value)

FROM employees;

-- With GROUP BY
SELECT dept_id,
  percentile_cont(0.5) WITHIN GROUP (ORDER BY salary) AS median_sal

Advanced Aggregates — STRING_AGG, JSON_AGG, MODE

String & JSON Aggregation

string_agg / json_agg
-- Concatenate with delimiter
SELECT dept_id, STRING_AGG(name, ', ') AS employees
FROM employees
GROUP BY dept_id;

-- With ORDER BY inside aggregate
SELECT dept_id, STRING_AGG(name, ', ' ORDER BY salary DESC) AS top_paid_first
FROM employees
GROUP BY dept_id;

-- JSON aggregation (PostgreSQL)
SELECT dept_id, JSON_AGG(JSON_BUILD_OBJECT('name', name, 'salary', salary))
FROM employees
GROUP BY dept_id;

-- JSON object aggregation
SELECT JSON_OBJECT_AGG(dept_id, cnt) FROM (
  SELECT dept_id, COUNT(*) AS cnt FROM employees GROUP BY dept_id
) t;
